==== .github/workflows/context-snapshot.yml ====
name: Repo Context Snapshot

on:
  workflow_dispatch: # dispara manualmente
  push:
    branches:
      - master
    paths:
      - ".github/workflows/context-snapshot.yml"

jobs:
  analyze-repo:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repo
        uses: actions/checkout@v3

      - name: Gerar contexto completo
        run: |
          find . -type f \
            -not -path "./.git/*" \
            -not -path "./node_modules/*" \
            \( -name "*.js" -o -name "*.ts" -o -name "*.json" -o -name "*.html" -o -name "*.css" \) \
            -exec sh -c 'echo "==== $1 ===="; cat "$1"' _ {} \; > full_context.txt

      - name: Upload contexto (artifact)
        uses: actions/upload-artifact@v3
        with:
          name: repo-context
          path: full_context.txt

==== .github/workflows/prompt-runner.yml ====
name: 🤖 Prompt Runner GPT-5-mini

on:
  push:
    branches:
      - main
      - master

jobs:
  run-prompt:
    runs-on: ubuntu-latest

    steps:
      - name: 🧩 Checkout
        uses: actions/checkout@v4

      - name: ⚙️ Setup environment
        run: |
          set -e
          mkdir -p .debug
          echo "OPENAI_API_KEY=${{ secrets.OPENAI_API_KEY }}" >> $GITHUB_ENV

      - name: 🧠 Build reduced context
        run: |
          set -e
          echo "🧠 Criando contexto leve..."
          MAX_SIZE=15000
          MAX_TOTAL=180000
          CURRENT_SIZE=0
          echo "" > full_context.txt

          for F in $(git diff --name-only HEAD~1 HEAD || git ls-files); do
            if [[ -f "$F" ]]; then
              SIZE=$(wc -c < "$F")
              if [ "$((CURRENT_SIZE + SIZE))" -gt "$MAX_TOTAL" ]; then
                echo "🚫 Limite total atingido ($MAX_TOTAL bytes)."
                break
              fi
              echo "==== $F ====" >> full_context.txt
              head -c "$MAX_SIZE" "$F" >> full_context.txt
              echo "\n" >> full_context.txt
              CURRENT_SIZE=$((CURRENT_SIZE + SIZE))
            fi
          done
          echo "📦 Contexto final: $(wc -c < full_context.txt) bytes"

      - name: 🧾 Generate AI Response (GPT-5-mini)
        id: ai
        run: |
          set -e
          COMMIT_MSG="$(git log -1 --pretty=%B | grep '^prompt:' | sed 's/^prompt: //')"
          echo "📝 Prompt detectado:"
          echo "$COMMIT_MSG"

          if [ -z "$COMMIT_MSG" ]; then
            echo "⚠️ Nenhum prompt encontrado no commit message. Abortando."
            exit 0
          fi

          REQUEST=$(jq -n \
            --arg model "gpt-5-mini-2025-08-07" \
            --arg prompt "$COMMIT_MSG" \
            --arg context "$(head -c 180000 full_context.txt)" \
            '{model: $model,
              messages: [
                {"role":"system","content":"Você é um assistente de programação chamado Cicada, que atua como uma IA especializada em desenvolvimento de software e automação para uma casa de apostas (bet house). Sempre retorne apenas JSON puro, sem caracteres especiais externos."},
                {"role":"user","content":($prompt + "\n\nContexto:\n" + $context)}
              ],
              max_completion_tokens: 3000
            }')

          echo "$REQUEST" > .debug/request.json
          RESPONSE=$(curl -s -X POST "https://api.openai.com/v1/chat/completions" \
            -H "Authorization: Bearer $OPENAI_API_KEY" \
            -H "Content-Type: application/json" \
            -d "$REQUEST")

          echo "$RESPONSE" > .debug/response_raw.txt
          jq -r '.choices[0].message.content' <<< "$RESPONSE" > ai_raw.txt
          echo "✅ Resposta bruta salva em ai_raw.txt"

      - name: 💾 Convert AI output to safe JSON
        run: |
          set -e
          echo "🧩 Processando saída da IA..."
          if jq empty ai_raw.txt 2>/dev/null; then
            # IA retornou JSON válido → usar diretamente
            cp ai_raw.txt ai_output.json
            echo "✅ AI output reconhecido como JSON válido."
          else
            # fallback → IA retornou apenas texto simples
            AI_CONTENT=$(jq -Rs '.' ai_raw.txt)
            echo "[{\"path\":\"README.md\",\"content\":$AI_CONTENT}]" > ai_output.json
            echo "⚙️ AI output tratado como texto simples."
          fi

          echo "📄 Pré-visualização do JSON final:"
          head -n 20 ai_output.json

      - name: 💾 Apply changes and commit
        run: |
          set -e
          echo "🧩 Aplicando alterações..."
          if jq -e '.changes' ai_output.json >/dev/null 2>&1; then
            echo "✅ Modo múltiplos arquivos detectado (campo .changes)"
            jq -c '.changes[]' ai_output.json | while read -r i; do
              FILE=$(echo "$i" | jq -r '.path')
              CONTENT=$(echo "$i" | jq -r '.content')
              mkdir -p "$(dirname "$FILE")"
              echo "$CONTENT" > "$FILE"
              echo "✅ Atualizado: $FILE"
            done
          else
            echo "⚙️ Modo simples (array de arquivos)."
            jq -c '.[]' ai_output.json | while read -r i; do
              FILE=$(echo "$i" | jq -r '.path')
              CONTENT=$(echo "$i" | jq -r '.content')
              mkdir -p "$(dirname "$FILE")"
              echo "$CONTENT" > "$FILE"
              echo "✅ Atualizado: $FILE"
            done
          fi

          git config user.name "prompt-bot"
          git config user.email "actions@github.com"
          if [[ -n "$(git status --porcelain)" ]]; then
            git add .
            git commit -m "🤖 AI applied changes for prompt"
            git push
            echo "🚀 Alterações enviadas com sucesso!"
          else
            echo "⚙️ Nenhuma alteração detectada."
          fi

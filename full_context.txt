==== .github/workflows/context-snapshot.yml ====
name: Repo Context Snapshot

on:
  workflow_dispatch: # dispara manualmente
  push:
    branches:
      - master
    paths:
      - ".github/workflows/context-snapshot.yml"

jobs:
  analyze-repo:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repo
        uses: actions/checkout@v3

      - name: Gerar contexto completo
        run: |
          find . -type f \
            -not -path "./.git/*" \
            -not -path "./node_modules/*" \
            \( -name "*.js" -o -name "*.ts" -o -name "*.json" -o -name "*.html" -o -name "*.css" \) \
            -exec sh -c 'echo "==== $1 ===="; cat "$1"' _ {} \; > full_context.txt

      - name: Upload contexto (artifact)
        uses: actions/upload-artifact@v3
        with:
          name: repo-context
          path: full_context.txt

==== .github/workflows/prompt-runner.yml ====
name: 🤖 Prompt Runner

on:
  push:
    branches:
      - master

jobs:
  run-prompt:
    runs-on: ubuntu-latest

    steps:
      - name: 🧩 Checkout
        uses: actions/checkout@v4

      - name: ⚙️ Setup environment
        run: |
          set -e
          mkdir -p .debug
          echo "OPENAI_API_KEY=${{ secrets.OPENAI_API_KEY }}" >> $GITHUB_ENV

      - name: 🧠 Build reduced context
        run: |
          set -e
          echo "🧠 Criando contexto leve..."
          MAX_SIZE=15000
          MAX_TOTAL=180000
          CURRENT_SIZE=0
          echo "" > full_context.txt

          for F in $(git diff --name-only HEAD~1 HEAD || git ls-files); do
            if [[ -f "$F" ]]; then
              SIZE=$(wc -c < "$F")
              if [ "$((CURRENT_SIZE + SIZE))" -gt "$MAX_TOTAL" ]; then
                echo "🚫 Limite total atingido ($MAX_TOTAL bytes)."
                break
              fi
              echo "==== $F ====" >> full_context.txt
              head -c "$MAX_SIZE" "$F" >> full_context.txt
              echo "\n" >> full_context.txt
              CURRENT_SIZE=$((CURRENT_SIZE + SIZE))
            fi
          done

          echo "📦 Contexto final: $(wc -c < full_context.txt) bytes"

      - name: 🧾 Generate AI Response (GPT-5)
        id: ai
        run: |
          set -e
          COMMIT_MSG="$(git log -1 --pretty=%B | grep '^prompt:' | sed 's/^prompt: //')"
          echo "📝 Prompt detectado:"
          echo "$COMMIT_MSG"

          if [ -z "$COMMIT_MSG" ]; then
            echo "⚠️ Nenhum prompt encontrado no commit message. Abortando."
            exit 0
          fi

          echo "🔄 Enviando para GPT-5..."
          REQUEST=$(jq -n \
            --arg model "gpt-5" \
            --arg prompt "$COMMIT_MSG" \
            --arg context "$(head -c 180000 full_context.txt)" \
            '{model: $model, messages: [
              {"role": "system", "content": "Você é um assistente que gera arquivos a partir de prompts de commit."},
              {"role": "user", "content": ($prompt + "\n\nContexto:\n" + $context)}
            ], max_tokens: 3000}')

          echo "📤 Requisição enviada:"
          echo "$REQUEST" | jq .

          RESPONSE=$(curl -s -w "\n\nHTTP_STATUS=%{http_code}\n" \
            -X POST "https://api.openai.com/v1/chat/completions" \
            -H "Authorization: Bearer ${{ secrets.OPENAI_API_KEY }}" \
            -H "Content-Type: application/json" \
            -d "$REQUEST")

          echo "$RESPONSE" > .debug/response_raw.txt
          echo "📬 Resposta bruta da API:"
          cat .debug/response_raw.txt

          HTTP_STATUS=$(grep "HTTP_STATUS=" .debug/response_raw.txt | cut -d= -f2)
          echo "🌐 Código HTTP: $HTTP_STATUS"

          if [ "$HTTP_STATUS" != "200" ]; then
            echo "🚨 Erro HTTP detectado. Abortando."
            exit 1
          fi

          # Extrai apenas o JSON útil da resposta
          echo "$RESPONSE" | grep -v "HTTP_STATUS=" > .debug/response.json

          jq -r '.choices[0].message.content' .debug/response.json > ai_output.txt || echo "" > ai_output.txt

          echo "✅ Trecho inicial da resposta GPT:"
          head -n 30 ai_output.txt

      - name: 💾 Apply changes
        run: |
          set -e
          if grep -q '"path"' ai_output.txt; then
            echo "🧩 Aplicando alterações..."
            jq -c '.[]' ai_output.txt | while read i; do
              FILE=$(echo "$i" | jq -r '.path')
              CONTENT=$(echo "$i" | jq -r '.content')
              mkdir -p "$(dirname "$FILE")"
              echo "$CONTENT" > "$FILE"
              echo "✅ Arquivo atualizado: $FILE"
            done
          else
            echo "⚠️ Nenhum JSON válido detectado. Resposta GPT ignorada."
            echo "👉 Resposta bruta salva em .debug/response_raw.txt"
          fi

      - name: 🚀 Commit & Push
        run: |
          set -e
          git config user.name "prompt-bot"
          git config user.email "actions@github.com"
          if [[ -n "$(git status --porcelain)" ]]; then
            git add .
            git commit -m "🤖 AI applied changes for prompt"
            git push
            echo "✅ Alterações enviadas!"
          else
            echo "⚙️ Nenhuma alteração detectada."
          fi
